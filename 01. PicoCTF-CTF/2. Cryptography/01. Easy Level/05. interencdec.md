````
import codecs

cipher = "wpjvJAM{jhlzhy_k3jy9wa3k_h47j6k69}"

# Shift -7 means shift +19 (since 26 - 7 = 19)
decoded = ""
for char in cipher:
    if char.isalpha():
        if char.isupper():
            decoded += chr((ord(char) - ord('A') - 7) % 26 + ord('A'))
        else:
            decoded += chr((ord(char) - ord('a') - 7) % 26 + ord('a'))
    else:
        decoded += char

print(decoded)
````
