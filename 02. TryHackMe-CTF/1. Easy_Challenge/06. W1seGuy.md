- ``ord()``:	Converts a character into a number
- ``^``	Performs XOR
- ``i % len(key)``: Repeats the key
- ``.hex()``: Encodes the encrypted result as hexadecimal

````py
import string
from pwn import *

charset = string.ascii_letters + string.digits

enc_flag = bytes.fromhex("172107183d7208260d3906113e2239375d29082e020738502c2f25330b18311d3353383111051130")

part_flag = b'THM{'

part_key = xor(enc_flag, part_flag)[:4]

#print(part_key)

for c in charset:
    key = part_key + c.encode()

    dec_flag = xor(enc_flag, key).decode()

    if dec_flag[-1] == '}':
        print(f"Key: {key.decode()}")
        print(f"Flag: {dec_flag}")
````
